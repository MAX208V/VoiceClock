    /** 已开启时报时则让服务重新计算下一次计划。 */
    private fun notifyServiceReload() {
        try {
            if (Settings.enabled(this)) {
                startService(Intent(this, ClockService::class.java).setAction(ClockService.RELOAD))
            }
        } catch (_: Throwable) {
            // 后台限制可能失败，不影响已保存的设置
        }
        // 刷新按钮文字（不影响开关状态）
        updateToggleButton()
    }
